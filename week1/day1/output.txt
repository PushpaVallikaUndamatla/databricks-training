    -- 62. Find employees hired in the same year
    SELECT e.*
    FROM Employee e
    JOIN (
        SELECT YEAR(hire_date) AS hire_year
        FROM Employee
        GROUP BY YEAR(hire_date)
        HAVING COUNT(*) > 1

    ON YEAR(e.hire_date) = y.hire_year;
